## ⚠️ 信令网关需要完善的功能

### 1. 🔴 SSRC管理（高优先级）

**当前问题**：
```php
// CommandDispatcher.php:515
$sdp .= "y=0100000001\r\n";  // ❌ 硬编码
```

**信令网关的职责**：
- ❌ 不应该在网关层生成SSRC（应该由gbvr-iot数据库管理）
- ✅ 网关应该**接收**SSRC（从Redis命令中传入）
- ✅ 网关负责将SSRC填入SDP

**解决方案**：

```php
// CommandDispatcher::handleStartLiveVideo()
private function handleStartLiveVideo(string $requestId, string $deviceId, 
    string $channelId, string $deviceIp, int $devicePort, array $params): array
{
    // ✅ 从params中获取SSRC（由gbvr-iot分配）
    $ssrc = $params['ssrc'] ?? null;
    
    if (!$ssrc) {
        return $this->errorResponse($requestId, "Missing SSRC from API");
    }
    
    // 分配ZLM端口（也应该从params传入）
    $zlmPort = $params['zlm_port'] ?? null;
    if (!$zlmPort) {
        return $this->errorResponse($requestId, "Missing ZLM port from API");
    }
    
    // 构建SDP
    $sdp = $this->buildInviteSdp(
        $this->config['media_server_ip'], 
        $zlmPort, 
        'Play', 
        'recvonly',
        $ssrc  // ← 使用传入的SSRC
    );
    
    // 发送INVITE...
}

private function buildInviteSdp(string $ip, int $port, string $sessionName, 
    string $mode, string $ssrc): string
{
    $serverId = $this->config['server_id'];
    
    $sdp = "v=0\r\n";
    $sdp .= "o={$serverId} 0 0 IN IP4 {$ip}\r\n";
    $sdp .= "s={$sessionName}\r\n";
    $sdp .= "c=IN IP4 {$ip}\r\n";
    $sdp .= "t=0 0\r\n";
    $sdp .= "m=video {$port} RTP/AVP 96 98 97\r\n";
    $sdp .= "a={$mode}\r\n";
    $sdp .= "a=rtpmap:96 PS/90000\r\n";
    $sdp .= "a=rtpmap:98 H264/90000\r\n";
    $sdp .= "a=rtpmap:97 MPEG4/90000\r\n";
    $sdp .= "y={$ssrc}\r\n";  // ✅ 使用传入的SSRC
    $sdp .= "f=\r\n";
    
    return $sdp;
}
```

**gbvr-iot的职责**：

```php
// gbvr-iot: DeviceChannelController.php
public function startLiveVideo(Request $request)
{
    $deviceId = $request->input('device_id');
    $channelId = $request->input('channel_id');
    
    // 1. 从数据库获取或创建通道
    $channel = DeviceChannel::firstOrCreate(
        ['device_id' => $deviceId, 'channel_id' => $channelId],
        ['ssrc' => $this->generateUniqueSsrc()]
    );
    
    // 2. 调用ZLM分配端口
    $zlmPort = $this->zlmService->openRtpServer($channel->stream_id, $channel->ssrc);
    
    // 3. 发送命令到Redis（信令网关订阅）
    Redis::publish('gb28181:commands', json_encode([
        'action' => 'start_live_video',
        'device_id' => $deviceId,
        'channel_id' => $channelId,
        'request_id' => uniqid(),
        'params' => [
            'ssrc' => $channel->ssrc,          // ← 传递SSRC
            'zlm_port' => $zlmPort,             // ← 传递ZLM端口
            'stream_id' => $channel->stream_id,
        ]
    ]));
    
    return response()->json(['success' => true]);
}

private function generateUniqueSsrc(): string
{
    // 数据库自增，确保唯一
    do {
        $ssrc = str_pad((string)rand(1000000000, 9999999999), 10, '0', STR_PAD_LEFT);
    } while (DeviceChannel::where('ssrc', $ssrc)->exists());
    
    return $ssrc;
}
```

```php
// 统一会话管理器
class SessionManager
{
    private array $sessions = [];
    
    /**
     * 创建会话
     */
    public function createSession(array $data): string
    {
        $sessionId = $data['call_id'] ?? uniqid('session_');
        
        $this->sessions[$sessionId] = [
            'session_id' => $sessionId,
            'call_id' => $data['call_id'] ?? '',
            'dialog_id' => $data['dialog_id'] ?? 0,
            'device_id' => $data['device_id'],
            'channel_id' => $data['channel_id'],
            'ssrc' => $data['ssrc'] ?? '',
            'stream_id' => $data['stream_id'] ?? '',
            'type' => $data['type'] ?? 'live',  // live/playback/talk
            'started_at' => time(),
            'last_activity' => time(),
        ];
        
        return $sessionId;
    }
    
    /**
     * 通过Call-ID查询
     */
    public function getByCallId(string $callId): ?array
    {
        return $this->sessions[$callId] ?? null;
    }
    
    /**
     * 通过Stream-ID查询
     */
    public function getByStreamId(string $streamId): ?array
    {
        foreach ($this->sessions as $session) {
            if ($session['stream_id'] === $streamId) {
                return $session;
            }
        }
        return null;
    }
    
    /**
     * 清理超时会话
     */
    public function cleanupTimeout(int $timeout = 3600): array
    {
        $now = time();
        $cleaned = [];
        
        foreach ($this->sessions as $sessionId => $session) {
            if ($now - $session['last_activity'] > $timeout) {
                $cleaned[] = $sessionId;
                unset($this->sessions[$sessionId]);
            }
        }
        
        return $cleaned;
    }
}
```

private function buildPlaybackSdp(string $ip, int $port, string $ssrc, 
    string $startTime, string $endTime): string
{
    $serverId = $this->config['server_id'];
    
    // 时间格式转换：2024-01-01T00:00:00 -> 20240101T000000Z
    $start = str_replace(['-', ':'], '', $startTime) . 'Z';
    $end = str_replace(['-', ':'], '', $endTime) . 'Z';
    
    $sdp = "v=0\r\n";
    $sdp .= "o={$serverId} 0 0 IN IP4 {$ip}\r\n";
    $sdp .= "s=Playback\r\n";
    $sdp .= "u={$channelId}:0\r\n";  // GB28181回放标识
    $sdp .= "c=IN IP4 {$ip}\r\n";
    $sdp .= "t={$start} {$end}\r\n";  // ← 关键：时间范围
    $sdp .= "m=video {$port} RTP/AVP 96 98 97\r\n";
    $sdp .= "a=recvonly\r\n";
    $sdp .= "a=rtpmap:96 PS/90000\r\n";
    $sdp .= "a=rtpmap:98 H264/90000\r\n";
    $sdp .= "a=rtpmap:97 MPEG4/90000\r\n";

namespace App\Services;

class SsrcService
{
    /**
     * 为通道分配唯一SSRC
     */
    public function allocate(string $deviceId, string $channelId): string
    {
        $channel = DeviceChannel::where([
            'device_id' => $deviceId,
            'channel_id' => $channelId
        ])->first();
        
        if ($channel && $channel->ssrc) {
            return $channel->ssrc;
        }
        
        // 生成唯一SSRC
        do {
            $ssrc = str_pad((string)rand(1000000000, 9999999999), 10, '0', STR_PAD_LEFT);
        } while (DeviceChannel::where('ssrc', $ssrc)->exists());
        
        return $ssrc;
    }
}

     * 发送命令到信令网关

    public function sendCommand(string $action, array $data): void
    {
        Redis::publish('gb28181:commands', json_encode([
            'action' => $action,
            ...$data,
            'request_id' => uniqid(),
            'timestamp' => time(),
        ]));
    }
    
    public function startLiveVideo(string $deviceId, string $channelId): void
    {
        $channel = $this->getOrCreateChannel($deviceId, $channelId);
        $zlmPort = $this->zlmService->openRtpServer($channel->stream_id, $channel->ssrc);
        
        $this->sendCommand('start_live_video', [
            'device_id' => $deviceId,
            'channel_id' => $channelId,
            'params' => [
                'ssrc' => $channel->ssrc,
                'zlm_port' => $zlmPort,
                'stream_id' => $channel->stream_id,
            ]
        ]);
    }
}
